\section*{Exercises}
\addcontentsline{toc}{section}{Exercises}
\subsection*{A. Theory}
\begin{exercise}\exerciseid{8.A1} Prove that $Ax=b$ has a solution if and only if $b\in\Img A$. \end{exercise}
\subsection*{B. By Hand}
\begin{exercise}\exerciseid{8.B1} Classify the system $\mat{1&1\\2&2}x=\mat{3\\6}$ as having no solution, one solution, or infinitely many solutions. \end{exercise}
\subsection*{C. Python / NumPy}
\begin{exercise}\exerciseid{8.C1} Use \texttt{np.linalg.lstsq} on a consistent system and an inconsistent system. Compare the residuals. \end{exercise}
\subsection*{D. Challenge}
\begin{exercise}\exerciseid{8.D1} Find all solutions of one underdetermined $2\times 3$ system and write them as $x_0+\Ker A$. \end{exercise}
