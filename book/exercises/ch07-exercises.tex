\section*{Exercises}
\addcontentsline{toc}{section}{Exercises}
\subsection*{A. Theory}
\begin{exercise}\exerciseid{7.A1} Prove that $\Ker A$ is a subspace of the domain and $\Img A$ is a subspace of the codomain. \end{exercise}
\subsection*{B. By Hand}
\begin{exercise}\exerciseid{7.B1} For $A=\mat{1&2&3\\0&1&1}$, find one nonzero vector in $\Ker A$. \end{exercise}
\subsection*{C. Python / NumPy}
\begin{exercise}\exerciseid{7.C1} Generate $A=UV$ with $U\in\R^{5\times 2}$ and $V\in\R^{2\times 4}$. Use \texttt{np.linalg.matrix\_rank} to estimate $\Rank A$. \end{exercise}
\subsection*{D. Challenge}
\begin{exercise}\exerciseid{7.D1} Construct a $3\times 4$ matrix with rank $2$ and explain why its nullity must be $2$. \end{exercise}
